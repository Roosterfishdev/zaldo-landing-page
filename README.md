```
├── site.webmanifest
├── favicon.ico

├── docs/                   Fuente legal en Markdown (no enlazada)
└── assets/

    ├── js/main.js
    ├── js/meta-pixel.js
    ├── aeonik-pro/         Regular, medium, bold, light

```
